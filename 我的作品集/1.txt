let audioCtx = null;
function playBerSound() {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    if (audioCtx.state === 'suspended') audioCtx.resume();

    const osc = audioCtx.createOscillator();
    const gainNode = audioCtx.createGain();

    osc.type = 'sine';
    osc.frequency.setValueAtTime(800, audioCtx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(300, audioCtx.currentTime + 0.1);

    gainNode.gain.setValueAtTime(0.5, audioCtx.currentTime);
    gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);

    osc.connect(gainNode);
    gainNode.connect(audioCtx.destination);

    osc.start();
    osc.stop(audioCtx.currentTime + 0.1);
  } catch (e) {}
}

document.addEventListener('click', (e) => {
  if (e.target.closest('a, button, .work-card, .contact-trigger, .method-item, .face-trigger')) {

const canvas = document.getElementById('bg-canvas');
const ctx = canvas.getContext('2d');
let particles = [];
let canvasW, canvasH;

function resizeCanvas(){
  canvasW = canvas.width = window.innerWidth;
  canvasH = canvas.height = window.innerHeight;
}
resizeCanvas();
window.addEventListener('resize', resizeCanvas);

const PARTICLE_COUNT = 60;
const MAX_DIST = 140;

class Particle {
  constructor(){
    this.x = Math.random() * canvasW;
    this.y = Math.random() * canvasH;
    this.vx = (Math.random() - 0.5) * 0.4;
    this.vy = (Math.random() - 0.5) * 0.4;
    this.r = Math.random() * 1.6 + 0.6;
  }
  update(){
    this.x += this.vx;
    this.y += this.vy;
    if (this.x < 0 || this.x > canvasW) this.vx *= -1;
    if (this.y < 0 || this.y > canvasH) this.vy *= -1;
  }
  draw(){
    ctx.beginPath();
    ctx.arc(this.x, this.y, this.r, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(200,255,46,0.5)';
    ctx.fill();
  }
}

for (let i = 0; i < PARTICLE_COUNT; i++) particles.push(new Particle());

let pMouseX = -9999, pMouseY = -9999;
document.addEventListener('mousemove', (e) => {
  pMouseX = e.clientX;
  pMouseY = e.clientY;
});

function drawLines(){
  for (let i = 0; i < particles.length; i++) {
    for (let j = i + 1; j < particles.length; j++) {
      const dx = particles[i].x - particles[j].x;
      const dy = particles[i].y - particles[j].y;
      const dist = Math.sqrt(dx * dx + dy * dy);
      if (dist < MAX_DIST) {
        const opacity = (1 - dist / MAX_DIST) * 0.18;
        ctx.beginPath();
        ctx.moveTo(particles[i].x, particles[i].y);
        ctx.lineTo(particles[j].x, particles[j].y);
        ctx.strokeStyle = `rgba(124,107,255,${opacity})`;
        ctx.lineWidth = 0.7;
        ctx.stroke();
      }
    }
    const dxm = particles[i].x - pMouseX;
    const dym = particles[i].y - pMouseY;
    const distM = Math.sqrt(dxm * dxm + dym * dym);
    if (distM < 180) {
      const opacity = (1 - distM / 180) * 0.35;
      ctx.beginPath();
      ctx.moveTo(particles[i].x, particles[i].y);
      ctx.lineTo(pMouseX, pMouseY);
      ctx.strokeStyle = `rgba(200,255,46,${opacity})`;
      ctx.lineWidth = 0.8;
      ctx.stroke();
    }
  }
}

function animateParticles(){
  ctx.clearRect(0, 0, canvasW, canvasH);
  particles.forEach(p => { p.update(); p.draw(); });
  drawLines();
  requestAnimationFrame(animateParticles);
}
animateParticles();

let eggTimers = [];

function resetEgg() {
  // 清除所有计时器
  eggTimers.forEach(clearTimeout);
  eggTimers = [];
  
  // 重置状态
  eggStage1.style.display = 'flex';
  eggStage2.style.display = 'none';
  eggStage3.style.display = 'none';
  
  // 停止并重置视频
  if (eggVideo) {
    eggVideo.pause();
    eggVideo.currentTime = 0;
  }

function openEgg() {
  resetEgg();
  eggModal.classList.add('open');
  document.body.classList.add('modal-open');
  document.body.style.overflow = 'hidden';

  // 阶段1 -> 阶段2 (等2秒)
  eggTimers.push(setTimeout(() => {
    if (eggModal.classList.contains('open')) {
      eggStage1.style.display = 'none';
      eggStage2.style.display = 'flex';
    }
  }, 2000));

  // 阶段2 -> 阶段3 (再等2秒)
  eggTimers.push(setTimeout(() => {
    if (eggModal.classList.contains('open')) {
      eggStage2.style.display = 'none';
      eggStage3.style.display = 'block';
      
      // 自动尝试播放视频
      if (eggVideo) {
        eggVideo.play().catch(e => console.log('视频自动播放被拦截，请手动点击播放'));
      }
    }
  }, 4000));
}

   12. 物理引擎 —— 全自动碰碰车 (空间宽敞版)

const { Engine, Runner, Bodies, Composite, Events } = Matter;

const container = document.getElementById('physicsContainer');
const pills = document.querySelectorAll('.skill-pill');

// 1. 初始化物理引擎
const engine = Engine.create();
const world = engine.world;
// 取消重力，让它们在无重力状态下自由漂浮碰撞
engine.gravity.y = 0; 

// 2. 建立墙壁
let containerWidth = container.clientWidth;
let containerHeight = container.clientHeight;

const wallOptions = { isStatic: true };
const ground = Bodies.rectangle(containerWidth / 2, containerHeight + 30, containerWidth, 60, wallOptions);
const leftWall = Bodies.rectangle(-30, containerHeight / 2, 60, containerHeight * 2, wallOptions);
const rightWall = Bodies.rectangle(containerWidth + 30, containerHeight / 2, 60, containerHeight * 2, wallOptions);
const ceiling = Bodies.rectangle(containerWidth / 2, -30, containerWidth, 60, wallOptions);

Composite.add(world, [ground, leftWall, rightWall, ceiling]);

// 3. 转化刚体
const pillBodies = [];
pills.forEach((pill) => {
  const w = pill.offsetWidth;
  const h = pill.offsetHeight;
  
  // 在容器内随机生成位置
  const x = Math.random() * (containerWidth - w) + w / 2;
  const y = Math.random() * (containerHeight - h) + h / 2;
  
  const body = Bodies.rectangle(x, y, w, h, {
    chamfer: { radius: h / 2 }, 
    restitution: 1,       // 完全弹性碰撞，动能不损失
    friction: 0,          // 无摩擦
    frictionAir: 0,       // 无空气阻力，永远匀速弹跳
    density: 0.001
  });
  
  // 给一个中等速度的初速度，让它们像碰碰车一样散开
  Matter.Body.setVelocity(body, {
    x: (Math.random() - 0.5) * 6,
    y: (Math.random() - 0.5) * 6
  });

  pillBodies.push(body);
  Composite.add(world, body);
});

// 4. 启动物理引擎并同步 DOM
const runner = Runner.create();
Runner.run(runner, engine);

Events.on(engine, 'afterUpdate', () => {
  pills.forEach((pill, index) => {
    const body = pillBodies[index];
    const x = body.position.x - pill.offsetWidth / 2;
    const y = body.position.y - pill.offsetHeight / 2;
    
    pill.style.transform = `translate(${x}px, ${y}px)`;
  });
});

  Matter.Body.setPosition(leftWall, { x: -30, y: newH / 2 });
  Matter.Body.setPosition(rightWall, { x: newW + 30, y: newH / 2 });
  Matter.Body.setPosition(ceiling, { x: newW / 2, y: -30 });
  Matter.Body.setVertices(ground, Matter.Bodies.rectangle(newW / 2, newH + 30, newW, 60).vertices);
  Matter.Body.setVertices(ceiling, Matter.Bodies.rectangle(newW / 2, -30, newW, 60).vertices);
});
